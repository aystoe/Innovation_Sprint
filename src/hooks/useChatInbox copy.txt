import { useEffect, useState } from 'react'
import { requireSupabase } from '../lib/supabase'
import { listConversations } from '../services/chatRepository'

type InboxItems = Awaited<ReturnType<typeof listConversations>>

type InboxState = {
  userId: string
  page: number
  attempt: number
  items: InboxItems
  error: string
}

export function useChatInbox(userId: string) {
  const [page, setPage] = useState(0)
  const [attempt, setAttempt] = useState(0)

  const [state, setState] = useState<InboxState>({
    userId: '',
    page: -1,
    attempt: -1,
    items: [],
    error: '',
  })

  useEffect(() => {
    const client = requireSupabase()
    let active = true
    let controller: AbortController | undefined
    let timer: ReturnType<typeof setTimeout> | undefined

    const refresh = async () => {
      controller?.abort()

      const request = new AbortController()
      controller = request

      try {
        const items = await listConversations(page, request.signal)

        if (active && !request.signal.aborted) {
          setState({
            userId,
            page,
            attempt,
            items,
            error: '',
          })
        }
      } catch {
        if (active && !request.signal.aborted) {
          setState({
            userId,
            page,
            attempt,
            items: [],
            error: 'Could not load conversations.',
          })
        }
      }
    }

    void refresh()

    const schedule = () => {
      clearTimeout(timer)
      timer = setTimeout(() => void refresh(), 150)
    }

    const channel = client
      .channel(`inbox:${userId}:${crypto.randomUUID()}`)
      .on(
        'postgres_changes',
        {
          event: 'INSERT',
          schema: 'public',
          table: 'messages',
        },
        schedule,
      )
      .on(
        'postgres_changes',
        {
          event: 'INSERT',
          schema: 'public',
          table: 'conversation_members',
          filter: `user_id=eq.${userId}`,
        },
        schedule,
      )
      .subscribe(status => {
        if (status === 'SUBSCRIBED') schedule()
      })

    const recover = () => {
      if (document.visibilityState !== 'hidden') {
        void refresh()
      }
    }

    const interval = setInterval(recover, 15000)

    window.addEventListener('focus', recover)
    window.addEventListener('online', recover)

    return () => {
      active = false
      controller?.abort()
      clearTimeout(timer)
      clearInterval(interval)
      void client.removeChannel(channel)
      window.removeEventListener('focus', recover)
      window.removeEventListener('online', recover)
    }
  }, [userId, page, attempt])

  const current =
    state.userId === userId &&
    state.page === page &&
    state.attempt === attempt

  return {
    items: current ? state.items : [],
    loading: !current,
    error: current ? state.error : '',
    page,
    setPage,
    retry: () => setAttempt(n => n + 1),
  }
}
